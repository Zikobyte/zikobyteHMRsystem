SECTION 2 — Hardware Requirements

ZMC Central Server

  • CPU: Intel Core i7 / Xeon E (8+ cores)
  • RAM: 16 GB DDR4 min (32 GB recommended)
  • Storage: 2x 1TB SSD RAID 1 + 4x 2TB HDD RAID 5
  • UPS: APC Smart-UPS 1500VA
  • OS: Ubuntu Server 22.04 LTS

Network Equipment

  • Switch: TP-Link TL-SG1016D (16-port Gigabit)
  • Cables: CAT6 Ethernet to each department
  • Patch panel + RJ45 wall plates per room
  • Wi-Fi AP: TP-Link EAP225 (mobile/tablet)

// --- LOGIN ---
app.post('/api/login', async (req, res) => {
  const { username, password } = req.body;
  try {
    const r = await db.query('SELECT * FROM users WHERE username = $1', [username]);
    if (!r.rows.length) return res.status(401).json({ error: 'Invalid credentials' });
    const user  = r.rows[0];
    const valid = await require('bcrypt').compare(password, user.password_hash);
    if (!valid) return res.status(401).json({ error: 'Invalid credentials' });
    const token = jwt.sign(
      { id: user.id, username: user.username, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: '8h' }
    );
    res.json({ token, role: user.role, username: user.username });
  } catch (e) { res.status(500).json({ error: 'Server error' }); }
});

// --- PATIENTS ---
app.get('/api/patients', auth, async (req, res) => {
  const r = await db.query('SELECT * FROM patients ORDER BY registration_date DESC');
  res.json(r.rows);
});

app.post('/api/patients', auth, async (req, res) => {
  const p = req.body;
  const r = await db.query(
    'INSERT INTO patients (name,date_of_birth,gender,phone_number,status,registered_by)'
    + ' VALUES ($1,$2,$3,$4,$5,$6) RETURNING *',
    [p.name, p.dateOfBirth, p.gender, p.phoneNumber, 'registered', req.user.username]
  );
  broadcast({ type: 'NEW_PATIENT', patient: r.rows[0] });
  res.json(r.rows[0]);
});

app.patch('/api/patients/:id', auth, async (req, res) => {
  const updates = req.body;
  const fields  = Object.keys(updates).map((k, i) => k + ' = $' + (i+2)).join(', ');
  const values  = [req.params.id, ...Object.values(updates)];
  const r = await db.query(
    'UPDATE patients SET ' + fields + ', updated_at = NOW() WHERE id = $1 RETURNING *',
    values
  );
  broadcast({ type: 'PATIENT_UPDATED', patient: r.rows[0] });
  res.json(r.rows[0]);
});

// --- NOTIFICATIONS ---
app.post('/api/notify', auth, async (req, res) => {
  const { targetRole, message, patientId, type } = req.body;
  await db.query(
    'INSERT INTO notifications (from_role,target_role,message,patient_id,type) VALUES ($1,$2,$3,$4,$5)',
    [req.user.role, targetRole, message, patientId, type]
  );
  broadcast({ type: 'NOTIFICATION', targetRole, message, patientId, from: req.user.role });
  res.json({ success: true });
});

// --- AUTH MIDDLEWARE ---
function auth(req, res, next) {
  const token = (req.headers.authorization || '').split(' ')[1];
  if (!token) return res.status(401).json({ error: 'No token' });
  try { req.user = jwt.verify(token, process.env.JWT_SECRET); next(); }
  catch { res.status(401).json({ error: 'Invalid token' }); }
}

app.listen(process.env.API_PORT, () => console.log('API running on ' + process.env.API_PORT));

// --- WEBSOCKET SERVER ---
const wss     = new WebSocket.Server({ port: process.env.WS_PORT });
const clients = new Map();

wss.on('connection', ws => {
  ws.on('message', msg => {
    try {
      const d = JSON.parse(msg);
      if (d.type === 'AUTH') {
        const u = jwt.verify(d.token, process.env.JWT_SECRET);
        clients.set(ws, { role: u.role, username: u.username });
      }
    } catch (e) { ws.close(); }
  });
  ws.on('close', () => clients.delete(ws));
});

function broadcast(msg) {
  const json = JSON.stringify(msg);
  wss.clients.forEach(client => {
    if (client.readyState !== WebSocket.OPEN) return;
    const info = clients.get(client);
    if (!msg.targetRole || !info || info.role === msg.targetRole)
      client.send(json);
  });
}

WebSocket Hook (React)

// src/hooks/useIntranetConnection.ts
import { useEffect, useRef, useCallback } from 'react';

const SERVER_WS = 'ws://192.168.1.1/ws';

export function useIntranetConnection(
  token: string,
  role: string,
  onMessage: (msg: any) => void
) {
  const ws             = useRef<WebSocket | null>(null);
  const reconnectTimer = useRef<ReturnType<typeof setTimeout>>();

  const connect = useCallback(() => {
    ws.current = new WebSocket(SERVER_WS);

    ws.current.onopen = () => {
      console.log('Connected to ZMC intranet server');
      ws.current?.send(JSON.stringify({ type: 'AUTH', token }));
    };

    ws.current.onmessage = (event) => {
      const data = JSON.parse(event.data);
      onMessage(data);
    };

    ws.current.onclose = () => {
      console.log('Disconnected — retrying in 3 seconds...');
      reconnectTimer.current = setTimeout(connect, 3000);
    };
  }, [token, onMessage]);

  useEffect(() => {
    connect();
    return () => {
      clearTimeout(reconnectTimer.current);
      ws.current?.close();
    };
  }, [connect]);

  const sendNotification = useCallback(
    (targetRole: string, message: string, patientId?: string) => {
      fetch('http://192.168.1.1/api/notify', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer ' + token,
        },
        body: JSON.stringify({ targetRole, message, patientId, type: 'PATIENT_FLOW' }),
      });
    },
    [token]
  );

  return { sendNotification };
}

Central API Config

// src/config/api.ts
export const API_BASE = 'http://192.168.1.1/api';
export const WS_URL   = 'ws://192.168.1.1/ws';

export async function apiRequest(path: string, options: RequestInit = {}): Promise<any> {
  const token = localStorage.getItem('zmc_token');
  const res = await fetch(API_BASE + path, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: 'Bearer ' + token } : {}),
      ...options.headers,
    },
  });
  if (!res.ok) throw new Error(await res.text());
  return res.json();
}

SECTION 9 — Security & Automated Backups

Network Security

  • Intranet physically isolated — no internet uplink
  • UFW firewall: only ports 80, 443, 22, 3001 open
  • PostgreSQL: only 192.168.1.x range allowed
  • Server in locked room

Application Security

  • JWT tokens expire after 8 hours
  • Passwords hashed with bcrypt
  • Role-based access per module
  • All actions logged with timestamp + username
